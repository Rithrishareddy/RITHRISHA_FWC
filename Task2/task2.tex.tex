\item\textcolor{cyan}{\textbf{Example 4 :}} In a right triangle ABC, right-angled at B,\\
if $\tan A = 1$, then verify that \\
\hspace*{1em} $2 \sin A \cos A = 1$.

\vspace{0.5em}
\noindent\textcolor{cyan}{\textbf{Solution :}} In $\triangle ABC$, $\tan A = \dfrac{BC}{AB} = 1$ \hfill

\vspace{-6em}
\begin{minipage}[t]{0.5\textwidth}
i.e.,\quad $BC = AB$ \\[0.5em]
Let $AB = BC = k$, where $k$ is a positive number. \\[0.5em]
Now,
\begin{align*}
AC &= \sqrt{AB^2 + BC^2} \\
&= \sqrt{(k)^2 + (k)^2} = k\sqrt{2}
\end{align*}

Therefore,\quad
$\sin A = \dfrac{BC}{AC} = \dfrac{k}{k\sqrt{2}} = \dfrac{1}{\sqrt{2}}$ \quad and \quad
$\cos A = \dfrac{AB}{AC} = \dfrac{k}{k\sqrt{2}} = \dfrac{1}{\sqrt{2}}$ \\[0.5em]

So,\quad $2 \sin A \cos A = 2 \left(\dfrac{1}{\sqrt{2}}\right)\left(\dfrac{1}{\sqrt{2}}\right) = 1$, which is the required value.
\end{minipage}

\item\textcolor{cyan}{\textbf{Example 5 :}} In $\triangle OPQ$, right-angled at P, \\
OP = 7 cm and OQ = PQ = 1 cm (see Fig. 8.12). \\
Determine the values of $\sin Q$ and $\cos Q$.

\vspace{-14em}
\noindent\textcolor{cyan}{\textbf{Solution :}} In $\triangle OPQ$, we have
\hspace{-7em}
\begin{minipage}[t]{0.65\textwidth}
\begin{align*}
    
OQ^2 &= OP^2 + PQ^2 \\
(1 + PQ)^2 &= OP^2 + PQ^2 \quad \text{(Why?)} \\
1 + PQ^2 + 2PQ &= OP^2 + PQ^2 \\
1 + 2PQ &= 7^2 \quad \text{(Why?)} \\
PQ &= 24\text{ cm and } OQ = 1 + PQ = 25\text{ cm}
\end{align*}

\vspace{-2em}
So, \quad $\sin Q = \dfrac{7}{25}$ and \quad $\cos Q = \dfrac{24}{25}$
\end{minipage}
